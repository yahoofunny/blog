六、编程题（每小题10分，共20分）
1、
code segment
  assume cs:code
start:
  mov bl,0
  mov cx,7
l1:
  mov ah,1
  int 21h
  cmp al,30h
  jl exit
  cmp al,31h
  jg exit
  shl bl,1
  sub al,30h
  and al,01h      ；本条指令可省
  add bl,al
  loop l1
output:
  mov dl,13
  mov ah,2
  int 21h
  mov dl,10
  mov ah,2
  int 21h
  mov dl,bl
  mov ah,2
  int 21h
exit:
  mov ah,4ch
  int 21h
code ends
  end start
代码段定义正确2分，loop循环正确2分，移位相加正确3分，输出正确3分。
2、
data segment
  buff db 16,?,16 dup(?),13,10,'$'
  inputs db 13,10,'input string:$'
  getc db 13,10,'input char:$'
  output db 13,10,'not found$'
data ends

code segment
  assume cs:code,ds:data,es:data
start:
  mov ax,data
  mov ds,ax
  mov ax,data
  mov es,ax

  lea dx,inputs
  mov ah,9
  int 21h

  lea dx,buff
  mov ah,10
  int 21h

  lea dx,getc
  mov ah,9
  int 21h

  mov ah,1
  int 21h     ;键盘输入字符ASCII码存放在al中

  lea di,buff+2
  mov cx,16
  cld
find:
  repne scasb
  jz yes
  jmp nof
yes:
  mov dl,13
  mov ah,2
  int 21h
  mov dl,10
  mov ah,2
  int 21h     ;回车换行

  mov bl,16
  sub bl,cl
  sub dl,1     ；应为SUB   BL，1  或DEC  BL

  mov dl,bl
  add dl,30h
  mov ah,2
  int 21h
  jmp exit
nof:
  lea dx,output
  mov ah,9
  int 21h
exit:
  mov ah,4ch
  int 21h
code ends
  end start

数据段定义正确1分，键盘缓冲区输入正确1分，字符输入正确1分，位置计算正确5分，
输出正确2分。
